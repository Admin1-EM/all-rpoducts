    <script>
        const SECRET_PASS = "vR8rdupxhnCYe4CLVoEHVQDmC";
        
        // Четем файла директно от същата папка в хостинга
        const JSON_URL = "products.json";

        let currentLang = 'bg';
        let cachedDatabase = null;

        const localizations = {
            bg: { welcome: "Здравейте! Въведете артикулен номер, за да проверя наличност и цена.", notfound: "Артикул с такъв код не беше намерен.", price: "Цена:", stock: "Наличност:", stock_ok: "В наличност", stock_no: "Няма в наличност", loading: "Зареждане на огромната база данни... Моля изчакайте малко." },
            'en-us': { welcome: "Hello! Enter an item code to check availability and price.", notfound: "Item with this code was not found.", price: "Price:", stock: "Stock:", stock_ok: "In stock", stock_no: "Out of stock", loading: "Loading database... Please wait." },
            ro: { welcome: "Bună! Introduceți un cod de articol pentru a verifica disponibilitatea și prețul.", notfound: "Articolul cu acest cod nu a fost găsit.", price: "Preț:", stock: "Stoc:", stock_ok: "În stoc", stock_no: "Nu este în stoc", loading: "Se încarcă baza de date... Vă rugăm să așteptați." }
        };

        async function checkPassword() {
            const input = document.getElementById('pass-input').value;
            if (input === SECRET_PASS) {
                document.getElementById('auth-container').style.display = 'none';
                document.getElementById('chat-container').style.display = 'flex';
                showBotMessage(localizations[currentLang].loading);

                try {
                    const response = await fetch(JSON_URL);
                    cachedDatabase = await response.json();
                    
                    document.getElementById('chat-messages').innerHTML = '';
                    showBotMessage(localizations[currentLang].welcome);
                } catch (error) {
                    showBotMessage("Грешка при зареждане на файла. Уверете се, че products.json е качен правилно.");
                    console.error(error);
                }
            } else {
                document.getElementById('error-msg').style.display = 'block';
            }
        }

        function setLanguage(lang) {
            currentLang = lang;
            document.querySelectorAll('.lang-selectors button').forEach(b => b.classList.remove('active'));
            document.getElementById(`btn-${lang === 'en-us' ? 'en' : lang}`).classList.add('active');
            if(cachedDatabase) {
                document.getElementById('chat-messages').innerHTML = '';
                showBotMessage(localizations[currentLang].welcome);
            }
        }

        function showBotMessage(text) {
            const container = document.getElementById('chat-messages');
            const msgDiv = document.createElement('div');
            msgDiv.className = 'msg bot';
            msgDiv.innerText = text;
            container.appendChild(msgDiv);
            container.scrollTop = container.scrollHeight;
        }

        function sendMessage() {
            const inputField = document.getElementById('user-input');
            const code = inputField.value.trim();
            if (!code || !cachedDatabase) return;

            const container = document.getElementById('chat-messages');
            const userDiv = document.createElement('div');
            userDiv.className = 'msg user';
            userDiv.innerText = code;
            container.appendChild(userDiv);
            inputField.value = '';

            const itemsList = cachedDatabase.items || cachedDatabase;
            const item = itemsList.find(i => i.code === code);
            
            setTimeout(() => {
                if (item) {
                    const cardDiv = document.createElement('div');
                    cardDiv.className = 'msg bot';
                    cardDiv.style.padding = '0';
                    cardDiv.style.background = 'transparent';
                    cardDiv.style.maxWidth = '100%';
                    
                    const name = item.name[currentLang] || item.name['bg'] || '';
                    const desc = item.short_description ? (item.short_description[currentLang] || item.short_description['bg'] || '') : '';
                    const isAvailable = item.availability && item.availability[0]?.status === 'available';
                    const stockText = isAvailable ? localizations[currentLang].stock_ok : localizations[currentLang].stock_no;
                    const stockClass = isAvailable ? 'stock-available' : 'stock-unavailable';
                    const imgUrl = item.images && item.images[0] ? item.images[0] : '';

                    cardDiv.innerHTML = `
                        <div class="product-card">
                            ${imgUrl ? `<img class="product-img" src="${imgUrl}" alt="product">` : ''}
                            <div class="product-details">
                                <div class="product-title">${name}</div>
                                <div class="product-price">${item.price.value} ${item.price.currency}</div>
                                <div class="product-stock ${stockClass}">${localizations[currentLang].stock}: ${stockText}</div>
                                <div class="product-desc">${desc.replace(/\n/g, '<br>')}</div>
                            </div>
                        </div>
                    `;
                    container.appendChild(cardDiv);
                } else {
                    showBotMessage(localizations[currentLang].notfound);
                }
                container.scrollTop = container.scrollHeight;
            }, 200);
        }
    </script>
